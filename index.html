왜 이렇게 해요?IL2CPP, ARM64, Target API가 뭐예요?

IL2CPP는 C# 코드를 폰이 바로 이해하는 기계어로 바꾸는 방식이에요. ARM64는 요즘 폰의 64비트 CPU로, 구글 플레이 필수 조건이에요. Target API는 "안드로이드 몇 버전 기준"이라는 표시로, Automatic이면 알아서 최신을 써요.

  5. 키 파일을 USB + 구글 드라이브 2곳에 복사

막히면스테이지가 없는 무한 게임이에요

왜 이렇게 해요?Firebase 켜기는 무슨 일을 해요?

두 가지를 해요. ① LAUNCHKIT_FIREBASE라는 스위치를 켜서 Mock 대신 진짜 Firebase 코드를 쓰게 해요. ② Assets/Plugins/Android에 Gradle 설정 파일 3개를 만들어서, EDM이 Firebase 안드로이드 라이브러리를 적어 넣을 자리를 만들어요. 게임 코드는 한 줄도 안 바뀌어요.

자주 묻는 질문

키트를 가져왔더니 원래 게임의 뒤로가기(Esc) 처리랑 겹쳐요

내 게임에 이미 Esc 키나 뒤로가기를 처리하는 코드가 있다면 그 코드를 지우고 BackButton.Push/Remove로 바꿔요. 둘 다 있으면 뒤로가기 한 번에 두 가지 일이 동시에 일어나요.

PlayerPrefs로 저장한 기존 데이터는 어떻게 되나요?

아직 출시 전이라면 그냥 SaveSystem으로 바꾸면 돼요. 이미 출시해서 유저가 있다면, 게임을 켤 때 PlayerPrefs에 값이 있으면 SaveSystem.Data로 옮기고 PlayerPrefs를 지우는 코드가 한 번 필요해요. 이건 AI에게 "PlayerPrefs의 best 값을 SaveSystem.Data.bestScore로 한 번만 옮기는 코드 만들어 줘"라고 부탁해 보세요.

가짜 광고 창이 떠 있을 때 뒤로가기를 누르면 왜 아무 일도 없나요?

진짜 광고는 광고 회사가 뒤로가기를 직접 처리해요. 그래서 키트는 광고가 떠 있는 동안 게임의 뒤로가기를 막아 둬요. 진짜 광고에서도 같은 순서로 동작하게 하려는 거예요.

Device Simulator에 내 폰 기종이 없어요

비슷한 화면 비율과 비슷한 카메라 구멍 위치의 기기를 골라도 충분해요. 가장 정확한 건 결국 진짜 폰에 설치해 보는 거예요.

Firebase를 켰더니 빌드가 Gradle 오류로 실패해요

먼저 Assets > External Dependency Manager > Android Resolver > Force Resolve 를 다시 하고 빌드해요. 그래도 안 되면 Console의 가장 위 빨간 줄을 복사해서 AI에게 물어봐요. 급하면 LaunchKit > Firebase 끄기 로 Mock으로 돌려서 빌드할 수 있어요.

Firebase를 켰는데 google-services.json이 없으면 어떻게 돼요?

빌드도 되고 게임도 멀쩡히 돌아가요. 키트가 폰에서 "설정 파일이 없네" 하고 Mock으로 돌아가요. 대신 기록은 Firebase로 가지 않아요. 점검 창의 Firebase 줄이 빨간색으로 알려 줘요.

AdMob을 켰더니 "Duplicate assembly" 경고가 나요

AdMob 플러그인이 유니티판 EDM을 하나 더 설치해서 Firebase의 구글판 EDM과 겹친 거예요. 점검 창 AdMob 줄의 정리하기 를 누르거나 LaunchKit > AdMob 켜기 를 다시 하면 하나만 남아요.

진짜 광고 단위 ID를 넣었는데 광고가 안 나와요

새로 만든 광고 단위는 몇 시간 걸려야 광고가 나오기 시작해요. 그리고 스토어에 출시해서 AdMob 앱과 스토어를 연결해야 광고가 제대로 채워져요. 그 전에는 테스트 광고로 확인해요.

SaveSystem.cs
using System;
using System.IO;
using UnityEngine;

// 저장과 불러오기를 맡습니다. 게임이 켜질 때 자동으로 불러오고,
// 앱이 백그라운드로 가거나 종료될 때 자동으로 저장합니다.
//
// 사용법
//   SaveSystem.Data.coins += 10;   // 값 바꾸기
//   SaveSystem.Save();             // 중요한 순간(보상 지급, 구매 직후)에는 바로 저장
//
// 저장 위치: Application.persistentDataPath/save.json
//   PC에서 확인하려면 메뉴 LaunchKit > 저장 폴더 열기
public static class SaveSystem
{
    public static SaveData Data { get; private set; } = new SaveData();

    public static string FilePath => Path.Combine(Application.persistentDataPath, "save.json");
    static string TempPath => FilePath + ".tmp";
    static string BackupPath => FilePath + ".bak";

    public static void Load()
    {
        Data = ReadFile(FilePath) ?? ReadFile(BackupPath) ?? new SaveData();
    }

    public static void Save()
    {
        try
        {
            // 저장 도중 앱이 꺼져도 파일이 깨지지 않도록
            // 1) 임시 파일에 다 쓴 다음  2) 기존 파일을 백업으로 돌리고  3) 임시 파일을 진짜 이름으로 바꿉니다.
            File.WriteAllText(TempPath, JsonUtility.ToJson(Data, true));
            if (File.Exists(FilePath))
            {
                if (File.Exists(BackupPath)) File.Delete(BackupPath);
                File.Move(FilePath, BackupPath);
            }
            File.Move(TempPath, FilePath);
        }
        catch (Exception e)
        {
            Debug.LogError($"[SaveSystem] 저장 실패: {e.Message}");
            LaunchKit.Crash?.RecordException(e);
        }
    }

    // 모든 저장 데이터를 지우고 처음 상태로 돌립니다. (설정 화면의 "초기화" 버튼용)
    public static void DeleteAll()
    {
        foreach (var path in new[] { FilePath, BackupPath, TempPath })
        {
            if (File.Exists(path)) File.Delete(path);
        }
        Data = new SaveData();
    }

    static SaveData ReadFile(string path)
    {
        try
        {
            if (!File.Exists(path)) return null;
            var data = JsonUtility.FromJson<SaveData>(File.ReadAllText(path));
            return data;
        }
        catch (Exception e)
        {
            // 파일이 깨졌으면 null을 돌려줘서 백업 파일을 읽게 합니다.
            Debug.LogWarning($"[SaveSystem] {Path.GetFileName(path)} 읽기 실패: {e.Message}");
            return null;
        }
    }
}

BackButton.cs
using System;
using System.Collections.Generic;
using UnityEngine;

// 안드로이드 뒤로가기 버튼 규칙
//   1. 열린 창(설정, 상점 등)이 있으면 가장 위의 창을 닫는다.
//   2. 열린 창이 없으면 "게임을 종료할까요?" 팝업을 띄운다.
//   3. 뒤로가기를 눌렀는데 아무 반응이 없으면 원스토어 검수에서 반려될 수 있다.
//
// 사용법
//   창을 열 때:  BackButton.Push(Close);
//   창을 닫을 때: BackButton.Remove(Close);   (Close는 그 창을 닫는 함수)
//   종료 팝업을 직접 만들었다면: BackButton.QuitHandler = 내팝업열기함수;
//   직접 만들지 않으면 키트의 기본 종료 팝업이 뜹니다.
public static class BackButton
{
    static readonly List<Action> stack = new List<Action>();
    static bool defaultPopupOpen;

    // 열린 창이 하나도 없을 때 불릴 함수입니다. 비워 두면 기본 종료 팝업이 뜹니다.
    public static Action QuitHandler;

    public static int OpenCount => stack.Count;

    public static void Push(Action onBack)
    {
        stack.Remove(onBack);
        stack.Add(onBack);
    }

    public static void Remove(Action onBack)
    {
        stack.Remove(onBack);
    }

    // 뒤로가기 버튼이 눌렸을 때 LaunchKitRunner가 부릅니다. (자동 테스트에서도 부릅니다)
    public static void Press()
    {
        if (MockAdOverlay.IsShowing) return; // 광고가 떠 있을 때는 광고가 뒤로가기를 처리합니다.

            stack.RemoveAt(stack.Count - 1);
            top();
            return;
        }

        if (QuitHandler != null) QuitHandler();
        else OpenDefaultQuitPopup();
    }

        Debug.Log("[LaunchKit] 게임 종료");

#endif
    }

        QuitHandler = null;

//   3. 배경 이미지처럼 화면 끝까지 꽉 채워야 하는 것은 SafeArea 밖(Canvas 바로 아래)에 둡니다.
//
// 에디터에서 확인하려면 Game 창 대신 Device Simulator 창(Window > General > Device Simulator)에서
// 노치가 있는 기기를 고르세요.
[RequireComponent(typeof(RectTransform))]
public class SafeAreaFitter : MonoBehaviour
{
    RectTransform rect;
    Rect lastSafeArea;
    Vector2Int lastScreen;

    void Awake()
    {
        rect = GetComponent<RectTransform>();
        Fit();
    }

    // 화면을 돌리거나 Device Simulator에서 기기를 바꾸면 안전 영역이 바뀌므로 매 프레임 확인합니다.
    void Update()
    {
        if (Screen.safeArea != lastSafeArea || Screen.width != lastScreen.x || Screen.height != lastScreen.y)
            Fit();
    }

    void Fit()
    {
        lastSafeArea = Screen.safeArea;
        lastScreen = new Vector2Int(Screen.width, Screen.height);
        if (Screen.width <= 0 || Screen.height <= 0) return;

        Vector2 min = lastSafeArea.position;
        Vector2 max = lastSafeArea.position + lastSafeArea.size;
        min.x /= Screen.width;
        min.y /= Screen.height;
        max.x /= Screen.width;
        max.y /= Screen.height;

        rect.anchorMin = min;
        rect.anchorMax = max;
        rect.offsetMin = Vector2.zero;
        rect.offsetMax = Vector2.zero;
    }
}

using System;
using System.Collections.Generic;
using Firebase;
using Firebase.Analytics;
using Firebase.Crashlytics;
using Firebase.Extensions;
using UnityEngine;

// 진짜 Firebase로 기록합니다. (3회차)
// 메뉴 LaunchKit > Firebase 켜기 를 누르면 Mock 대신 이 파일이 쓰입니다.
//
// 안전장치
//  - Firebase가 준비되기 전(앱을 켜고 1~2초)에 보낸 기록은 모아 두었다가 준비되면 보냅니다.
//  - google-services.json이 없거나 준비에 실패하면 게임을 멈추지 않고 Mock으로 돌아갑니다.
//  - 에디터에서는 Firebase가 기록을 보내지 않으므로 Mock처럼 Console에만 찍습니다.
public class FirebaseService : IAnalyticsService, ICrashReporter
{
    enum State { Starting, Ready, Fallback }

    // 지금 상태를 다른 곳(테스트, 디버그 화면)에서 볼 수 있게 열어 둡니다.
    public static string StatusText { get; private set; } = "준비 중";

    const int MaxPending = 200;

    State state = State.Starting;
    bool started;
    readonly MockAnalytics mockAnalytics = new MockAnalytics();
    readonly MockCrashReporter mockCrash = new MockCrashReporter();
    readonly List<Action> pending = new List<Action>();

    // LaunchKit이 Analytics, Crash 두 번 부르지만 준비는 한 번만 합니다.
    public void Initialize()
    {
        if (started) return;
        started = true;

        if (Application.isEditor)
        {
            UseFallback("에디터에서는 Firebase로 보내지 않아요. 폰에서 확인하세요.", warn: false);
            return;
        }

        try
        {
            FirebaseApp.CheckAndFixDependenciesAsync().ContinueWithOnMainThread(task =>
            {
                if (task.IsFaulted || task.IsCanceled)
                {
                    UseFallback("준비 실패: " + task.Exception?.GetBaseException().Message);
                    return;
                }
                if (task.Result != DependencyStatus.Available)
                {
                    UseFallback("구글 플레이 서비스 문제: " + task.Result);
                    return;
                }

                try
                {
                    // google-services.json이 빠졌으면 여기서 오류가 납니다.
                    var app = FirebaseApp.DefaultInstance;
                    Crashlytics.ReportUncaughtExceptionsAsFatal = true;
                    FirebaseAnalytics.SetAnalyticsCollectionEnabled(true);

                    state = State.Ready;
                    StatusText = "Firebase 연결됨 (" + app.Options.ProjectId + ")";
                    Debug.Log("<color=#81c784>[Firebase]</color> " + StatusText + ", 대기 중이던 기록 " + pending.Count + "개 보냄");
                    Flush();
                }
                catch (Exception e)
                {
                    UseFallback("google-services.json을 확인하세요: " + e.Message);
                }
            });
        }
        catch (Exception e)
        {
            UseFallback("준비 실패: " + e.Message);
        }
    }

    void UseFallback(string reason, bool warn = true)
    {
        state = State.Fallback;
        StatusText = "Mock 사용 중 - " + reason;
        if (warn) Debug.LogWarning("[Firebase] " + StatusText);
        else Debug.Log("<color=#81c784>[Firebase]</color> " + StatusText);
        mockAnalytics.Initialize();
        mockCrash.Initialize();
        Flush();
    }

    void Flush()
    {
        var list = pending.ToArray();
        pending.Clear();
        foreach (var action in list) action();
    }

    // 상태에 따라 진짜로 보내거나, Mock으로 찍거나, 준비될 때까지 모아 둡니다.
    void Run(Action real, Action mock)
    {
        switch (state)
        {
            case State.Ready:
                try { real(); }
                catch (Exception e) { Debug.LogWarning("[Firebase] 보내기 실패: " + e.Message); }
                break;
            case State.Fallback:
                mock();
                break;
            default:
                if (pending.Count < MaxPending) pending.Add(() => Run(real, mock));
                break;
        }
    }

    // ---------- 행동 기록 (Analytics) ----------

    public void LogEvent(string eventName, params AnalyticsParam[] parameters)
    {
        Run(() =>
        {
            var list = new Parameter[parameters.Length];
            for (int i = 0; i < parameters.Length; i++)
            {
                var p = parameters[i];
                list[i] = p.Value switch
                {
                    long l => new Parameter(p.Name, l),
                    double d => new Parameter(p.Name, d),
                    _ => new Parameter(p.Name, p.Value?.ToString() ?? ""),
                };
            }
            FirebaseAnalytics.LogEvent(eventName, list);
        },
        () => mockAnalytics.LogEvent(eventName, parameters));
    }

    public void SetUserProperty(string name, string value)
    {
        Run(() => FirebaseAnalytics.SetUserProperty(name, value),
            () => mockAnalytics.SetUserProperty(name, value));
    }

    // ---------- 오류 보고 (Crashlytics) ----------

    public void Log(string message)
    {
        Run(() => Crashlytics.Log(message), () => mockCrash.Log(message));
    }

    public void RecordException(Exception exception)
    {
        Run(() => Crashlytics.LogException(exception), () => mockCrash.RecordException(exception));
    }

    public void SetKey(string key, string value)
    {
        Run(() => Crashlytics.SetCustomKey(key, value), () => mockCrash.SetKey(key, value));
    }
}

using UnityEngine;

// 진짜 AdMob 광고입니다. (4회차)
// 메뉴 LaunchKit > AdMob 켜기 를 누르면 Mock 대신 이 파일이 쓰입니다.

// 순서

//  3. 광고를 보여 주고 닫히면 다음 광고를 다시 받아 둡니다. 실패하면 조금 기다렸다 다시 시도합니다.
public class AdMobService : IAdService
{
    RewardedAd rewarded;
    InterstitialAd interstitial;
    BannerView banner;
    bool started;          // AdMob을 켰는지
    bool bannerWanted;     // 게임이 배너를 보여 달라고 했는지
    int rewardedRetry, interstitialRetry;

    public bool IsRewardedReady => rewarded != null && rewarded.CanShowAd();
    public bool IsInterstitialReady => interstitial != null && interstitial.CanShowAd();

    public void Initialize()
    {
        // 광고 콜백을 Unity 메인 스레드에서 받습니다. 키트도 한 번 더 메인 스레드로 옮겨서 이중으로 안전하게 해요.
        MobileAds.RaiseAdEventsOnUnityMainThread = true;
        Debug.Log("<color=#aed581>[AdMob]</color> 개인정보 동의 확인 중..." +
                  (AdMobIds.ForceTestAds ? " (테스트 광고 모드)" : ""));

        // 동의(UMP) 콜백은 Unity 메인 스레드가 아닌 곳에서 올 수 있어서, 키트가 메인 스레드로 옮겨 실행합니다.
        // (폰에서 확인: 옮기지 않으면 다음 단계가 실행되지 않고 광고가 영영 안 나왔어요)
        var request = new ConsentRequestParameters();
        ConsentInformation.Update(request, updateError => LaunchKitRunner.RunOnMainThread(() =>
        {
            if (updateError != null)
                Debug.LogWarning("[AdMob] 동의 정보 확인 실패: " + updateError.Message);

            // 필요한 사람에게만 동의 창이 뜹니다. 한국 유저는 보통 바로 넘어가요.
            ConsentForm.LoadAndShowConsentFormIfRequired(formError => LaunchKitRunner.RunOnMainThread(() =>
            {
                if (formError != null)
                    Debug.LogWarning("[AdMob] 동의 창 오류: " + formError.Message);
                Debug.Log("<color=#aed581>[AdMob]</color> 동의 확인 끝. 광고 요청 가능: " + ConsentInformation.CanRequestAds());
                if (ConsentInformation.CanRequestAds()) StartAds();
            }));
        }));

        // 지난번에 이미 동의했다면 확인을 기다리지 않고 바로 시작합니다.
        if (ConsentInformation.CanRequestAds()) StartAds();

        started = true;
        MobileAds.Initialize(status => LaunchKitRunner.RunOnMainThread(() =>
        {
            Debug.Log("<color=#aed581>[AdMob]</color> 준비 완료. 광고를 불러오는 중...");
            LoadRewarded();
            LoadInterstitial();
            if (bannerWanted) CreateBanner();
        }));
    }

    // ---------- 보상형 ----------

    void LoadRewarded()
    {
        rewarded?.Destroy();
        rewarded = null;
        RewardedAd.Load(AdMobIds.RewardedId, new AdRequest(), (ad, error) => LaunchKitRunner.RunOnMainThread(() =>
        {
            if (error != null || ad == null)
            {
                Debug.LogWarning("[AdMob] 보상형 광고 불러오기 실패: " + error?.GetMessage());
                Retry(LoadRewarded, ref rewardedRetry);
                return;
            }
            rewardedRetry = 0;
            rewarded = ad;
            Debug.Log("<color=#aed581>[AdMob]</color> 보상형 광고 준비됨");
        }));
    }

    public void ShowRewarded(Action<bool> onFinished)
    {
        if (!IsRewardedReady) { onFinished?.Invoke(false); return; }

        bool earned = false, done = false;
        var ad = rewarded;
        void Finish(bool watched)
        {
            if (done) return;
            done = true;
            KitGUI.BlockGameInput(true);   // 광고를 닫은 손가락이 게임 버튼을 누르지 않게
            KitGUI.BlockGameInput(false);
            onFinished?.Invoke(watched);
            LoadRewarded();
        }
        ad.OnAdFullScreenContentClosed += () => LaunchKitRunner.RunOnMainThread(() => Finish(earned));
        ad.OnAdFullScreenContentFailed += err => LaunchKitRunner.RunOnMainThread(() =>
        {
            Debug.LogWarning("[AdMob] 보상형 광고 표시 실패: " + err.GetMessage());
            Finish(false);
        });
        ad.Show(reward => earned = true);
    }

        interstitial = null;
        InterstitialAd.Load(AdMobIds.InterstitialId, new AdRequest(), (ad, error) => LaunchKitRunner.RunOnMainThread(() =>
        {
            if (error != null || ad == null)
            {
                Debug.LogWarning("[AdMob] 전면 광고 불러오기 실패: " + error?.GetMessage());

        var ad = interstitial;

            done = true;
            KitGUI.BlockGameInput(true);
            KitGUI.BlockGameInput(false);
            onClosed?.Invoke();
            LoadInterstitial();
        }
        ad.OnAdFullScreenContentClosed += () => LaunchKitRunner.RunOnMainThread(Finish);
        ad.OnAdFullScreenContentFailed += err => LaunchKitRunner.RunOnMainThread(() =>
        {
            Debug.LogWarning("[AdMob] 전면 광고 표시 실패: " + err.GetMessage());

    void CreateBanner()

        banner = new BannerView(AdMobIds.BannerId, size, AdPosition.Bottom);
        banner.OnBannerAdLoadFailed += err => Debug.LogWarning("[AdMob] 배너 불러오기 실패: " + err.GetMessage());
        banner.LoadAd(new AdRequest());
        if (!bannerWanted) banner.Hide();
    }

    // ---------- 개인정보 설정 ----------

    public bool PrivacyOptionsRequired =>
        ConsentInformation.PrivacyOptionsRequirementStatus == PrivacyOptionsRequirementStatus.Required;

    public void ShowPrivacyOptions(Action onClosed)
    {
        ConsentForm.ShowPrivacyOptionsForm(error => LaunchKitRunner.RunOnMainThread(() =>
        {
            if (error != null) Debug.LogWarning("[AdMob] 개인정보 설정 창 오류: " + error.Message);
            onClosed?.Invoke();
        }));
    }

    // 실패하면 10초, 20초, 40초 ... 최대 2분까지 늘려 가며 다시 시도합니다.
    static void Retry(Action load, ref int count)
    {
        count++;
        RetryAfter(load, Mathf.Min(10 * (1 << Mathf.Min(count - 1, 4)), 120));
    }

    static async void RetryAfter(Action load, int seconds)
    {
        await Task.Delay(seconds * 1000);
        if (Application.isPlaying) load();
    }
}

TapGame.cs (예시 게임)

[키트] 라고 적힌 주석이 내 게임에 옮겨 쓸 부분이에요.

TapGame.cs
using TMPro;
using UnityEngine;
using UnityEngine.UI;

// 출시 키트 사용 예시 게임: 10초 동안 버튼을 최대한 많이 누르면 그만큼 코인을 받습니다.
// 게임 자체는 중요하지 않습니다. "어디서 LaunchKit을 부르는지"를 보는 용도입니다.
// 주석에 [키트]라고 적힌 줄이 내 게임에도 그대로 옮겨 쓸 부분입니다.
public class TapGame : MonoBehaviour
{
    [Header("타이틀 화면")]
    public GameObject titlePanel;
    public TMP_Text coinsText;
    public TMP_Text bestText;
    public Button playButton;
    public Button settingsButton;

    [Header("플레이 화면")]
    public GameObject playPanel;
    public TMP_Text scoreText;
    public TMP_Text timerText;
    public Button tapButton;

    [Header("결과 화면")]
    public GameObject resultPanel;
    public TMP_Text resultText;
    public Button doubleButton;
    public TMP_Text doubleLabel;
    public Button okButton;

    [Header("설정 창")]
    public GameObject settingsPanel;
    public Button soundButton;
    public TMP_Text soundLabel;
    public Button resetButton;
    public Button privacyButton;
    public Button closeSettingsButton;

    [Header("종료 확인 창")]
    public GameObject quitPanel;
    public Button quitYesButton;
    public Button quitNoButton;

    [Header("규칙")]
    public float roundSeconds = 10f;
    public int interstitialEveryRounds = 3; // 몇 판마다 전면 광고를 시도할지

    public int Score { get; private set; }
    public int Earned { get; private set; }
    public bool Playing { get; private set; }

    float timeLeft;
    bool doubled;
    bool firstRound;
    AudioSource audioSource;
    AudioClip tapClip;

    void Start()
    {
        playButton.onClick.AddListener(StartRound);
        tapButton.onClick.AddListener(Tap);
        doubleButton.onClick.AddListener(WatchAdForDouble);
        okButton.onClick.AddListener(FinishResult);
        settingsButton.onClick.AddListener(OpenSettings);
        soundButton.onClick.AddListener(ToggleSound);
        resetButton.onClick.AddListener(ResetSave);
        privacyButton.onClick.AddListener(OpenPrivacyOptions);
        closeSettingsButton.onClick.AddListener(CloseSettings);
        quitYesButton.onClick.AddListener(BackButton.QuitGame);
        quitNoButton.onClick.AddListener(CloseQuit);

        // [키트] 열린 창이 없을 때 뒤로가기를 누르면 내가 만든 종료 확인 창을 띄웁니다.
        BackButton.QuitHandler = OpenQuit;

        audioSource = gameObject.AddComponent<AudioSource>();
        tapClip = MakeBeep();

        ShowTitle();
    }

    void Update()
    {
        if (!Playing) return;
        timeLeft -= Time.deltaTime;
        timerText.text = Mathf.CeilToInt(Mathf.Max(0f, timeLeft)).ToString();
        if (timeLeft <= 0f) EndRound();
    }

    // ---------------- 타이틀 ----------------

    void ShowTitle()
    {
        titlePanel.SetActive(true);
        playPanel.SetActive(false);
        resultPanel.SetActive(false);
        coinsText.text = $"Coins: {SaveSystem.Data.coins}";
        bestText.text = $"Best: {SaveSystem.Data.bestScore}";
        LaunchKit.ShowBanner();                        // [키트] 배너는 플레이 중이 아닌 화면에서만
        LaunchKit.Crash.SetKey("screen", "title");     // [키트] 튕겼을 때 어느 화면이었는지 남기기
    }

    // ---------------- 플레이 ----------------

    public void StartRound()
    {
        Score = 0;
        timeLeft = roundSeconds;
        Playing = true;
        firstRound = !SaveSystem.Data.tutorialDone;
        SaveSystem.Data.playCount++;

        titlePanel.SetActive(false);
        playPanel.SetActive(true);
        scoreText.text = "0";
        LaunchKit.HideBanner();
        LaunchKit.Crash.SetKey("screen", "play");

        // [키트] 판 시작 기록. 첫 판이면 튜토리얼 시작도 기록합니다.
        if (firstRound) LaunchKit.Analytics.LogEvent(GameEvents.TutorialBegin);
        LaunchKit.Analytics.LogEvent(GameEvents.LevelStart,
            new AnalyticsParam(GameEvents.ParamLevelName, "tap_round"));

        // [키트] 플레이 중 뒤로가기 = 이번 판 포기
        BackButton.Push(CancelRound);
    }

    public void Tap()
    {
        if (!Playing) return;
        Score++;
        scoreText.text = Score.ToString();
        audioSource.PlayOneShot(tapClip);
    }

    void CancelRound()
    {
        if (!Playing) return;
        Playing = false;
        LaunchKit.Analytics.LogEvent(GameEvents.LevelEnd,
            new AnalyticsParam(GameEvents.ParamLevelName, "tap_round"),
            new AnalyticsParam(GameEvents.ParamSuccess, 0));
        ShowTitle();
    }

    public void EndRound()
    {
        if (!Playing) return;
        Playing = false;
        BackButton.Remove(CancelRound);

        Earned = Score;
        doubled = false;
        if (Score > SaveSystem.Data.bestScore) SaveSystem.Data.bestScore = Score;

        // [키트] 판 종료 기록: 점수가 들어 있어서 "유저들이 보통 몇 점을 내는지" 알 수 있습니다.
        LaunchKit.Analytics.LogEvent(GameEvents.LevelEnd,
            new AnalyticsParam(GameEvents.ParamLevelName, "tap_round"),
            new AnalyticsParam(GameEvents.ParamSuccess, 1),
            new AnalyticsParam(GameEvents.ParamScore, Score));
        if (firstRound)
        {
            SaveSystem.Data.tutorialDone = true;
            LaunchKit.Analytics.LogEvent(GameEvents.TutorialComplete);
        }

        playPanel.SetActive(false);
        resultPanel.SetActive(true);
        LaunchKit.Crash.SetKey("screen", "result");
        RefreshResult();
    }

    // ---------------- 결과 ----------------

    void RefreshResult()
    {
        resultText.text = $"Score {Score}\n+{Earned} coins";
        doubleButton.gameObject.SetActive(!doubled && Earned > 0);
    }

    void LateUpdate()
    {
        // [키트] 광고가 아직 준비 안 됐으면 버튼을 잠가 둡니다. 누르고 나서 아무 일도 안 생기면 유저가 답답해합니다.
        if (resultPanel.activeSelf && doubleButton.gameObject.activeSelf)
        {
            bool ready = LaunchKit.Ads.IsRewardedReady;
            doubleButton.interactable = ready;
            doubleLabel.text = ready ? "x2 Coins (AD)" : "Loading AD...";
        }
    }

    public void WatchAdForDouble()
    {
        // [키트] 보상은 반드시 onRewarded 안에서만 줍니다. 광고를 끝까지 본 경우에만 불립니다.
        LaunchKit.ShowRewardedAd("result_double", onRewarded: () =>
        {
            doubled = true;
            Earned *= 2;
            RefreshResult();
        });
    }

    public void FinishResult()
    {
        SaveSystem.Data.coins += Earned;
        SaveSystem.Save(); // [키트] 보상을 준 직후에는 바로 저장합니다.
        LaunchKit.Analytics.LogEvent(GameEvents.EarnCurrency,
            new AnalyticsParam(GameEvents.ParamCurrencyName, "coin"),
            new AnalyticsParam(GameEvents.ParamValue, Earned));

        resultPanel.SetActive(false);

        // [키트] 몇 판마다 전면 광고. 광고가 안 떠도 onClosed는 항상 불리므로 그냥 다음 화면으로 가면 됩니다.
        if (SaveSystem.Data.playCount % interstitialEveryRounds == 0)
            LaunchKit.ShowInterstitialAd("after_result", ShowTitle);
        else
            ShowTitle();
    }

    // ---------------- 설정 ----------------

    public void OpenSettings()
    {
        settingsPanel.SetActive(true);
        RefreshSound();
        // [키트] 동의가 필요한 지역(유럽 등)의 유저에게만 "개인정보 설정" 버튼을 보여 줍니다. AdMob 정책이에요.
        privacyButton.gameObject.SetActive(LaunchKit.Ads.PrivacyOptionsRequired);
        BackButton.Push(CloseSettings); // [키트] 창을 열면 뒤로가기로 닫히게 등록
    }

    public void CloseSettings()
    {
        settingsPanel.SetActive(false);
        BackButton.Remove(CloseSettings); // [키트] 창을 닫으면 등록 해제
    }

    void ToggleSound()
    {
        SoundSettings.SetSound(!SoundSettings.SoundOn);
        RefreshSound();
    }

    void RefreshSound()
    {
        soundLabel.text = SoundSettings.SoundOn ? "Sound: ON" : "Sound: OFF";
    }

    void OpenPrivacyOptions()
    {
        LaunchKit.Ads.ShowPrivacyOptions(null); // [키트] 구글 동의 창을 다시 띄워서 고를 수 있게
    }

    void ResetSave()
    {
        SaveSystem.DeleteAll();
        SoundSettings.Apply();
        RefreshSound();
        CloseSettings();
        ShowTitle();
    }

    // ---------------- 종료 확인 ----------------

    public void OpenQuit()
    {
        quitPanel.SetActive(true);
        BackButton.Push(CloseQuit); // 종료 창에서 뒤로가기를 한 번 더 누르면 창이 닫힙니다.
    }

    public void CloseQuit()
    {
        quitPanel.SetActive(false);
        BackButton.Remove(CloseQuit);
    }

    // 효과음 파일 없이 코드로 "삑" 소리를 만듭니다.
    static AudioClip MakeBeep()
    {
        const int rate = 44100;
        var samples = new float[rate / 20];
        for (int i = 0; i < samples.Length; i++)
        {
            float fade = 1f - (float)i / samples.Length;
            samples[i] = Mathf.Sin(2f * Mathf.PI * 880f * i / rate) * 0.3f * fade;
        }
        var clip = AudioClip.Create("beep", samples.Length, 1, rate, false);
        clip.SetData(samples, 0);
        return clip;
    }
}
